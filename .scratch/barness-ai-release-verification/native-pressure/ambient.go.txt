func udpCondition(dir string) {

 receivers:=[]*net.UDPConn{}
 senders:=[]*net.UDPConn{}
 data:=make([]byte,(16<<10)-64)
 var wg sync.WaitGroup
 var mu sync.Mutex
 reported:=int64(0)
 errors:=0
 causes:=map[string]int{}
 for range pairs {

 }
 wg.Wait()
